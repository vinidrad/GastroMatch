/*
GASTROMATCH - CRIACAO DO BANCO E DAS TABELAS
Este arquivo cria o banco apenas se ele ainda não existir e cria as tabelas.
Não insere usuários, cursos, receitas ou outros dados.
Não apaga o banco existente.
*/

USE master;
GO

IF DB_ID('GastroMatch') IS NULL
    CREATE DATABASE GastroMatch;
GO

USE GastroMatch;
GO

-- =========================================
-- USUARIOS
-- =========================================
CREATE TABLE usuarios (
    id INT IDENTITY(1,1) PRIMARY KEY,
    foto_perfil VARCHAR(255),
    nome VARCHAR(150) NOT NULL,
    email VARCHAR(150) NOT NULL UNIQUE,
    telefone VARCHAR(20),
    senha VARCHAR(255) NOT NULL,
    bio VARCHAR(150),
    cnpj VARCHAR(18),
    StatusCnpj VARCHAR(20) NOT NULL DEFAULT 'NaoSolicitado',
    certificado VARCHAR(255),
    StatusCertificado VARCHAR(20) NOT NULL DEFAULT 'NaoSolicitado',
    chef BIT NOT NULL DEFAULT 0,
    restaurante BIT NOT NULL DEFAULT 0,
    cliente BIT NOT NULL DEFAULT 1,
    CONSTRAINT CK_usuarios_tipo
        CHECK (
            (CAST(chef AS INT) +
             CAST(restaurante AS INT) +
             CAST(cliente AS INT)) = 1
        )
);
GO

-- =========================================
-- ATIVIDADES: 1 = CURSO, 2 = RECEITA
-- =========================================
CREATE TABLE atividades (
    id INT IDENTITY(1,1) PRIMARY KEY,
    nome VARCHAR(200) NOT NULL,
    descricao VARCHAR(MAX),
    sobre VARCHAR(MAX),
    tipo INT NOT NULL,
    categoria VARCHAR(100) NOT NULL,
    valor DECIMAL(10,2) NOT NULL DEFAULT 0.00,
    tempo_curso INT,
    imagem VARCHAR(255),
    fk_usuario_id INT NOT NULL,
    data_cadastro DATETIME NOT NULL DEFAULT GETDATE(),
    CONSTRAINT fk_atividade_usuario
        FOREIGN KEY (fk_usuario_id) REFERENCES usuarios(id),
    CONSTRAINT chk_atividade_tipo CHECK (tipo IN (1, 2)),
    CONSTRAINT chk_atividade_valor CHECK (valor >= 0)
);
GO

-- =========================================
-- AULAS
-- =========================================
CREATE TABLE aulas (
    id INT IDENTITY(1,1) PRIMARY KEY,
    fk_atividade_id INT NOT NULL,
    titulo VARCHAR(200) NOT NULL,
    video VARCHAR(255) NOT NULL,
    ordem INT NOT NULL,
    data_cadastro DATETIME NOT NULL DEFAULT GETDATE(),
    CONSTRAINT fk_aula_atividade
        FOREIGN KEY (fk_atividade_id)
        REFERENCES atividades(id)
        ON DELETE CASCADE
);
GO

-- =========================================
-- LIKES E DESLIKES DAS AULAS
-- avaliacao = 1: like | avaliacao = -1: deslike
-- =========================================
CREATE TABLE avaliacoes_aulas (
    id INT IDENTITY(1,1) PRIMARY KEY,
    fk_aula_id INT NOT NULL,
    fk_usuario_id INT NOT NULL,
    avaliacao INT NOT NULL,
    data_cadastro DATETIME NOT NULL DEFAULT GETDATE(),
    CONSTRAINT fk_avaliacao_aula
        FOREIGN KEY (fk_aula_id) REFERENCES aulas(id) ON DELETE CASCADE,
    CONSTRAINT fk_avaliacao_aula_usuario
        FOREIGN KEY (fk_usuario_id) REFERENCES usuarios(id),
    CONSTRAINT chk_avaliacao_aula_tipo
        CHECK (avaliacao IN (1, -1)),
    CONSTRAINT uq_avaliacao_aula_usuario
        UNIQUE (fk_aula_id, fk_usuario_id)
);
GO


-- =========================================
-- COMPRAS
-- =========================================
CREATE TABLE compras (
    id INT IDENTITY(1,1) PRIMARY KEY,
    fk_usuario_id INT NOT NULL,
    fk_atividade_id INT NOT NULL,
    valor_pago DECIMAL(10,2) NOT NULL,
    data_compra DATETIME NOT NULL DEFAULT GETDATE(),
    status INT NOT NULL DEFAULT 1,
    CONSTRAINT fk_compra_usuario
        FOREIGN KEY (fk_usuario_id) REFERENCES usuarios(id),
    CONSTRAINT fk_compra_atividade
        FOREIGN KEY (fk_atividade_id) REFERENCES atividades(id),
    CONSTRAINT chk_compra_status CHECK (status IN (1, 2, 3)),
    CONSTRAINT chk_compra_valor CHECK (valor_pago >= 0)
);
GO

-- =========================================
-- AVALIACOES
-- =========================================
CREATE TABLE avaliacoes (
    id INT IDENTITY(1,1) PRIMARY KEY,
    fk_usuario_id INT NOT NULL,
    fk_atividade_id INT NOT NULL,
    nota DECIMAL(2,1) NOT NULL,
    comentario VARCHAR(MAX),
    data DATETIME NOT NULL DEFAULT GETDATE(),
    CONSTRAINT chk_nota CHECK (nota >= 0 AND nota <= 5),
    CONSTRAINT fk_avaliacao_usuario
        FOREIGN KEY (fk_usuario_id) REFERENCES usuarios(id),
    CONSTRAINT fk_avaliacao_atividade
        FOREIGN KEY (fk_atividade_id) REFERENCES atividades(id),
    CONSTRAINT uq_usuario_atividade UNIQUE (fk_usuario_id, fk_atividade_id)
);
GO

-- =========================================
-- PERGUNTAS - COMUNIDADE
-- =========================================
CREATE TABLE perguntas (
    id INT IDENTITY(1,1) PRIMARY KEY,
    fk_usuario_id INT NOT NULL,
    assunto VARCHAR(200) NOT NULL,
    pergunta VARCHAR(MAX) NOT NULL,
    data DATETIME NOT NULL DEFAULT GETDATE(),
    CONSTRAINT fk_pergunta_usuario
        FOREIGN KEY (fk_usuario_id) REFERENCES usuarios(id)
);
GO

-- =========================================
-- SALVOS / FAVORITOS
-- =========================================
CREATE TABLE salvos (
    id INT IDENTITY(1,1) PRIMARY KEY,
    fk_usuario_id INT NOT NULL,
    fk_atividade_id INT NOT NULL,
    data_salvo DATETIME NOT NULL DEFAULT GETDATE(),
    CONSTRAINT fk_produto_salvo_usuario
        FOREIGN KEY (fk_usuario_id)
        REFERENCES usuarios(id)
        ON DELETE CASCADE,
    CONSTRAINT fk_produto_salvo_atividade
        FOREIGN KEY (fk_atividade_id)
        REFERENCES atividades(id)
        ON DELETE CASCADE,
    CONSTRAINT uq_usuario_atividade_salvo
        UNIQUE (fk_usuario_id, fk_atividade_id)
);
GO

-- =========================================
-- RESPOSTAS - COMUNIDADE
-- =========================================
CREATE TABLE respostas (
    id INT IDENTITY(1,1) PRIMARY KEY,
    fk_pergunta_id INT NOT NULL,
    fk_usuario_id INT NOT NULL,
    resposta VARCHAR(MAX) NOT NULL,
    data DATETIME NOT NULL DEFAULT GETDATE(),
    CONSTRAINT fk_resposta_pergunta
        FOREIGN KEY (fk_pergunta_id)
        REFERENCES perguntas(id)
        ON DELETE CASCADE,
    CONSTRAINT fk_resposta_usuario
        FOREIGN KEY (fk_usuario_id) REFERENCES usuarios(id)
);
GO
